// Array хранит элементы, их количество и размер выделенной памяти.
type Array struct {
        data     []string
        size     int
        capacity int
}

func (array *Array) Size() int {
        return array.size
}

func (array *Array) Push(value string) {
        array.Insert(array.size, value)
}

// При нехватке места увеличивает массив и вставляет значение по индексу.
func (array *Array) Insert(index int, value string) {
        if index < 0 || index > array.size {
                panic("index out of range")
        }
        if array.size == array.capacity {
                newCapacity := 4
                if array.capacity > 0 {
                        newCapacity = array.capacity * 2
                }
                newData := make([]string, newCapacity)
                copy(newData, array.data)
                array.data = newData
                array.capacity = newCapacity
        }
        for i := array.size; i > index; i-- {
                array.data[i] = array.data[i-1]
        }
        array.data[index] = value
        array.size++
}

func (array *Array) Get(index int) string {
        if index < 0 || index >= array.size {
                panic("index out of range")
        }
        return array.data[index]
}

func (array *Array) Set(index int, value string) {
        if index < 0 || index >= array.size {
                panic("index out of range")
        }
        array.data[index] = value
}

// Удаляет элемент и сдвигает оставшиеся значения влево.
func (array *Array) Delete(index int) string {
        value := array.Get(index)
        for i := index; i+1 < array.size; i++ {
                array.data[i] = array.data[i+1]
        }
        array.size--
        array.data[array.size] = ""
        return value
}

func (array *Array) Values() []string {
        result := make([]string, array.size)
        copy(result, array.data[:array.size])
        return result
}

// doubleNode хранит ссылки на предыдущий и следующий узлы.
type doubleNode struct {
        value string
        prev  *doubleNode
        next  *doubleNode
}

type DoubleList struct {
        head *doubleNode
        tail *doubleNode
        size int
}

func (list *DoubleList) Size() int {
        return list.size
}

func (list *DoubleList) at(index int) *doubleNode {
        if index < 0 || index >= list.size {
                panic("index out of range")
        }
        node := list.head
        for i := 0; i < index; i++ {
                node = node.next
        }
        return node
}

// Вставляет узел и обновляет связи с соседними узлами.
func (list *DoubleList) Insert(index int, value string) {
        if index < 0 || index > list.size {
                panic("index out of range")
        }
        node := &doubleNode{value: value}
        if index == list.size {
                node.prev = list.tail
                if list.tail == nil {
                        list.head = node
                } else {
                        list.tail.next = node
                }
                list.tail = node
        } else {
                after := list.at(index)
                node.next = after
                node.prev = after.prev
                if after.prev == nil {
                        list.head = node
                } else {
                        after.prev.next = node
                }
                after.prev = node
        }
        list.size++
}

func (list *DoubleList) Get(index int) string {
        return list.at(index).value
}

// Удаляет узел, сохраняя прямые и обратные связи списка.
func (list *DoubleList) Delete(index int) string {
        node := list.at(index)
        if node.prev == nil {
                list.head = node.next
        } else {
                node.prev.next = node.next
        }
        if node.next == nil {
                list.tail = node.prev
        } else {
                node.next.prev = node.prev
        }
        list.size--
        return node.value
}

func (list *DoubleList) Find(value string) int {
        index := 0
        for node := list.head; node != nil; node = node.next {
                if node.value == value {
                        return index
                }
                index++
        }
        return -1
}

// Читает список от начала или от конца.
func (list *DoubleList) Values(reverse bool) []string {
        result := []string{}
        if reverse {
                for node := list.tail; node != nil; node = node.prev {
                        result = append(result, node.value)
                }
        } else {
                for node := list.head; node != nil; node = node.next {
                        result = append(result, node.value)
                }
        }
        return result
}

import "sort"

type hashEntry struct {
        key, value string
        used       bool
}
type HashTable struct {
        cuckoo        bool
        chains        [][]hashEntry
        first, second []hashEntry
        count         int
        seed          uint64
}

func NewHashTable(cuckoo bool) *HashTable {
        h := &HashTable{cuckoo: cuckoo}
        if cuckoo {
                h.first = make([]hashEntry, 11)
                h.second = make([]hashEntry, 11)
        } else {
                h.chains = make([][]hashEntry, 11)
        }
        return h
}
func (h *HashTable) Size() int { return h.count }

func hashCode(s string, salt uint64) uint64 {
        h := uint64(1469598103934665603) ^ salt
        for i := 0; i < len(s); i++ {
                h ^= uint64(s[i])
                h *= 1099511628211
        }
        h ^= h >> 33
        h *= 0xff51afd7ed558ccd
        h ^= h >> 33
        return h
}
func (h *HashTable) bucket(key string, table int) int {
        capacity := len(h.chains)
        if h.cuckoo {
                capacity = len(h.first)
        }
        salt := h.seed
        if table == 1 {
                salt += 0x9e3779b97f4a7c15
        }
        return int(hashCode(key, salt) % uint64(capacity))
}
func (h *HashTable) slot(key string, table int) *hashEntry {
        if table == 0 {
                return &h.first[h.bucket(key, 0)]
        }
        return &h.second[h.bucket(key, 1)]
}
func (h *HashTable) Capacity() int {
        if h.cuckoo {
                return len(h.first) * 2
        }
        return len(h.chains)
}
func (h *HashTable) Get(key string) (string, bool) {
        if h.cuckoo {
                for i := 0; i < 2; i++ {
                        p := h.slot(key, i)
                        if p.used && p.key == key {
                                return p.value, true
                        }
                }
        } else {
                for _, e := range h.chains[h.bucket(key, 0)] {
                        if e.key == key {
                                return e.value, true
                        }
                }
        }
        return "", false
}
func (h *HashTable) entries() []hashEntry {
        out := make([]hashEntry, 0, h.count)
        if h.cuckoo {
                for _, p := range h.first {
                        if p.used {
                                out = append(out, p)
                        }
                }
                for _, p := range h.second {
                        if p.used {
                                out = append(out, p)
                        }
                }
        } else {
                for _, b := range h.chains {
                        out = append(out, b...)
                }
        }
        return out
}
func (h *HashTable) insertRaw(e hashEntry) *hashEntry {
        table := 0
        for i := 0; i < 4*len(h.first); i++ {
                p := h.slot(e.key, table)
                if !p.used {
                        *p = e
                        h.count++
                        return nil
                }
                e, *p = *p, e
                table ^= 1
        }
        return &e
}
func (h *HashTable) rebuild(entries []hashEntry, capacity int) {
        for {
                h.first = make([]hashEntry, capacity)
                h.second = make([]hashEntry, capacity)
                h.count = 0
                h.seed++
                failed := false
                for i, e := range entries {
                        if displaced := h.insertRaw(e); displaced != nil {
                                retry := append([]hashEntry{}, entries[i+1:]...)
                                retry = append(retry, *displaced)
                                retry = append(retry, h.entries()...)
                                entries = retry
                                failed = true
                                break
                        }
                }
                if !failed {
                        return
                }
                capacity = capacity*2 + 1
        }
}
func (h *HashTable) Put(key, value string) bool {
        if h.cuckoo {
                for i := 0; i < 2; i++ {
                        p := h.slot(key, i)
                        if p.used && p.key == key {
                                p.value = value
                                return false
                        }
                }
                if (h.count+1)*2 > h.Capacity() {
                        all := append(h.entries(), hashEntry{key, value, true})
                        h.rebuild(all, len(h.first)*2+1)
                        return true
                }
                if displaced := h.insertRaw(hashEntry{key, value, true}); displaced != nil {
                        all := append(h.entries(), *displaced)
                        h.rebuild(all, len(h.first)*2+1)
                }
                return true
        }
        index := h.bucket(key, 0)
        for i := range h.chains[index] {
                if h.chains[index][i].key == key {
                        h.chains[index][i].value = value
                        return false
                }
        }
        if (h.count+1)*10 > len(h.chains)*9 {
                old := h.entries()
                h.chains = make([][]hashEntry, len(h.chains)*2+1)
                h.count = 0
                for _, e := range old {
                        h.Put(e.key, e.value)
                }
        }
        index = h.bucket(key, 0)
        h.chains[index] = append(h.chains[index], hashEntry{key, value, true})
        h.count++
        return true
}
func (h *HashTable) Delete(key string) bool {
        if h.cuckoo {
                for i := 0; i < 2; i++ {
                        p := h.slot(key, i)
                        if p.used && p.key == key {
                                *p = hashEntry{}
                                h.count--
                                return true
                        }
                }
        } else {
                index := h.bucket(key, 0)
                b := h.chains[index]
                for i, e := range b {
                        if e.key == key {
                                copy(b[i:], b[i+1:])
                                b[len(b)-1] = hashEntry{}
                                h.chains[index] = b[:len(b)-1]
                                h.count--
                                return true
                        }
                }
        }
        return false
}
func (h *HashTable) Keys() []string {
        keys := make([]string, 0, h.count)
        for _, e := range h.entries() {
                keys = append(keys, e.key)
        }
        sort.Strings(keys)
        return keys
}

const (
        binaryHeader   = "LR3G"
        maxItems       = 1_000_000
        maxStringBytes = 16 * 1024 * 1024
)

// Lab1Data groups the structures from Lab 1 into one binary snapshot.
type Lab1Data struct {
        Array  Array
        Single SingleList
        Double DoubleList
        Queue  Queue
        Stack  Stack
}

func (data *Lab1Data) MarshalBinary() ([]byte, error) {
        out := []byte(binaryHeader)
        groups := [][]string{
                data.Array.Values(), data.Single.Values(), data.Double.Values(false),
                data.Queue.Values(), data.Stack.Values(),
        }
        for _, group := range groups {
                if len(group) > maxItems {
                        return nil, errors.New("too many items")
                }
                out = binary.LittleEndian.AppendUint32(out, uint32(len(group)))
                for _, value := range group {
                        if len(value) > maxStringBytes {
                                return nil, errors.New("string too large")
                        }
                        out = binary.LittleEndian.AppendUint32(out, uint32(len(value)))
                        out = append(out, value...)
                }
        }
        return out, nil
}

func readCount(in *bytes.Reader, limit uint32) (uint32, error) {
        var value uint32
        if err := binary.Read(in, binary.LittleEndian, &value); err != nil {
                return 0, err
        }
        if value > limit {
                return 0, errors.New("invalid length")
        }
        return value, nil
}

func readStrings(in *bytes.Reader) ([]string, error) {
        count, err := readCount(in, maxItems)
        if err != nil {
                return nil, err
        }
        values := []string{}
        for i := uint32(0); i < count; i++ {
                size, err := readCount(in, maxStringBytes)
                if err != nil {
                        return nil, err
                }
                if uint64(size) > uint64(in.Len()) {
                        return nil, io.ErrUnexpectedEOF
                }
                value := make([]byte, size)
                if _, err := io.ReadFull(in, value); err != nil {
                        return nil, err
                }
                values = append(values, string(value))
        }
        return values, nil
}

func (data *Lab1Data) UnmarshalBinary(raw []byte) error {
        if !bytes.HasPrefix(raw, []byte(binaryHeader)) {
                return errors.New("invalid binary header")
        }
        in := bytes.NewReader(raw[len(binaryHeader):])
        var groups [5][]string
        for i := range groups {
                values, err := readStrings(in)
                if err != nil {
                        return fmt.Errorf("group %d: %w", i, err)
                }
                groups[i] = values
        }
        if in.Len() != 0 {
                return errors.New("unexpected data after stack")
        }

        var next Lab1Data
        for _, value := range groups[0] {
                next.Array.Push(value)
        }
        for _, value := range groups[1] {
                next.Single.Insert(next.Single.Size(), value)
        }
        for _, value := range groups[2] {
                next.Double.Insert(next.Double.Size(), value)
        }
        for _, value := range groups[3] {
                next.Queue.Push(value)
        }
        for i := len(groups[4]) - 1; i >= 0; i-- {
                next.Stack.Push(groups[4][i])
        }
        *data = next
        return nil
}

func (h *HashTable) MarshalText() ([]byte, error) {
        var out strings.Builder
        method := "chain"
        if h.cuckoo {
                method = "cuckoo"
        }
        out.WriteString("LR3T " + method + "\n")
        for _, key := range h.Keys() {
                value, _ := h.Get(key)
                out.WriteString(hex.EncodeToString([]byte(key)))
                out.WriteByte('\t')
                out.WriteString(hex.EncodeToString([]byte(value)))
                out.WriteByte('\n')
        }
        return []byte(out.String()), nil
}

func (h *HashTable) UnmarshalText(raw []byte) error {
        lines := strings.Split(strings.TrimSuffix(string(raw), "\n"), "\n")
        header := strings.TrimSuffix(lines[0], "\r")
        if header != "LR3T chain" && header != "LR3T cuckoo" {
                return errors.New("invalid text header")
        }
        next := NewHashTable(header == "LR3T cuckoo")
        for _, line := range lines[1:] {
                keyHex, valueHex, found := strings.Cut(strings.TrimSuffix(line, "\r"), "\t")
                if !found {
                        return errors.New("missing key/value separator")
                }
                key, err := hex.DecodeString(keyHex)
                if err != nil {
                        return err
                }
                value, err := hex.DecodeString(valueHex)
                if err != nil {
                        return err
                }
                if !next.Put(string(key), string(value)) {
                        return errors.New("duplicate hash key")
                }
        }
        *h = *next
        return nil
}

// Очередь хранит начало и конец; первым удаляется первый добавленный элемент.
type queueNode struct {
        value string
        next  *queueNode
}

type Queue struct {
        head *queueNode
        tail *queueNode
}

func (queue *Queue) Push(value string) {
        node := &queueNode{value: value}
        if queue.tail == nil {
                queue.head = node
        } else {
                queue.tail.next = node
        }
        queue.tail = node
}

func (queue *Queue) Get() string {
        if queue.head == nil {
                panic("queue is empty")
        }
        return queue.head.value
}

func (queue *Queue) Pop() string {
        value := queue.Get()
        queue.head = queue.head.next
        if queue.head == nil {
                queue.tail = nil
        }
        return value
}

func (queue *Queue) Values() []string {
        result := []string{}
        for node := queue.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

// singleNode хранит значение и ссылку на следующий узел.
type singleNode struct {
        value string
        next  *singleNode
}

type SingleList struct {
        head *singleNode
        tail *singleNode
        size int
}

func (list *SingleList) Size() int {
        return list.size
}

// Находит узел по индексу, последовательно проходя список.
func (list *SingleList) at(index int) *singleNode {
        if index < 0 || index >= list.size {
                panic("index out of range")
        }
        node := list.head
        for i := 0; i < index; i++ {
                node = node.next
        }
        return node
}

// Вставляет новый узел в начало или после предыдущего узла.
func (list *SingleList) Insert(index int, value string) {
        if index < 0 || index > list.size {
                panic("index out of range")
        }
        node := &singleNode{value: value}
        if index == 0 {
                node.next = list.head
                list.head = node
                if list.tail == nil {
                        list.tail = node
                }
        } else {
                before := list.tail
                if index < list.size {
                        before = list.at(index - 1)
                }
                node.next = before.next
                before.next = node
                if node.next == nil {
                        list.tail = node
                }
        }
        list.size++
}

func (list *SingleList) Get(index int) string {
        return list.at(index).value
}

// Удаляет узел и соединяет соседние элементы списка.
func (list *SingleList) Delete(index int) string {
        if index < 0 || index >= list.size {
                panic("index out of range")
        }
        var node *singleNode
        if index == 0 {
                node = list.head
                list.head = node.next
                if list.head == nil {
                        list.tail = nil
                }
        } else {
                before := list.at(index - 1)
                node = before.next
                before.next = node.next
                if before.next == nil {
                        list.tail = before
                }
        }
        list.size--
        return node.value
}

func (list *SingleList) Find(value string) int {
        index := 0
        for node := list.head; node != nil; node = node.next {
                if node.value == value {
                        return index
                }
                index++
        }
        return -1
}

func (list *SingleList) Values() []string {
        result := []string{}
        for node := list.head; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}

// Стек хранит вершину; первым удаляется последний добавленный элемент.
type stackNode struct {
        value string
        next  *stackNode
}

type Stack struct {
        top *stackNode
}

func (stack *Stack) Push(value string) {
        stack.top = &stackNode{value: value, next: stack.top}
}

func (stack *Stack) Get() string {
        if stack.top == nil {
                panic("stack is empty")
        }
        return stack.top.value
}

func (stack *Stack) Pop() string {
        value := stack.Get()
        stack.top = stack.top.next
        return value
}

func (stack *Stack) Values() []string {
        result := []string{}
        for node := stack.top; node != nil; node = node.next {
                result = append(result, node.value)
        }
        return result
}
